import java.util.Locale;

/**
 * Lenient enum parsing so the API accepts "Under Review", "UNDER_REVIEW", "under-review"
 * and "underreview" for the same value.
 */
public final class EnumLookup {

    private EnumLookup() {
    }

    private static String normalize(String s) {
        return s == null ? "" : s.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]", "");
    }

    /** Returns null for null/blank input; throws IllegalArgumentException for an unknown value. */
    public static <E extends Enum<E> & LabeledEnum> E parse(Class<E> type, String input) {
        if (input == null || input.isBlank()) {
            return null;
        }
        String wanted = normalize(input);
        for (E candidate : type.getEnumConstants()) {
            if (normalize(candidate.name()).equals(wanted) || normalize(candidate.getLabel()).equals(wanted)) {
                return candidate;
            }
        }
        throw new IllegalArgumentException("Unknown " + type.getSimpleName() + ": " + input);
    }
}

/**
 * One piece of instructor feedback (or a student's clarification question) about a component.
 *
 * The text of a feedback record is never edited in place. When the instructor changes their
 * instructions, a new record is created whose previousFeedback points at the old one and the old
 * one becomes SUPERSEDED, so every earlier version stays on file.
 */
@Getter
@Setter
@Entity
@Table(name = "feedback_items")
public class Feedback {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "project_id", nullable = false)
    private Project project;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "component_id", nullable = false)
    private ProjectComponent component_id;

    @Enumerated(EnumType.STRING)
    @Column(length = 20, nullable = false)
    private FeedbackType type = FeedbackType.FEEDBACK;

    @Column(name = "feedback_text", columnDefinition = "TEXT", nullable = false)
    private String text;

    /** Who gave the feedback (usually the adviser). */
    private String givenBy;

    /** The date the feedback was received; the student can set it when recording older feedback. */
    private LocalDate receivedOn;

    @Enumerated(EnumType.STRING)
    @Column(length = 20, nullable = false)
    private FeedbackStatus status = FeedbackStatus.OPEN;

    private int versionNumber = 1;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "previous_feedback_id")
    private Feedback previous_feedback_id;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "recorded_by_id")
    private User recorded_by_id;

    private Instant createdAt;

    private Instant resolvedAt;

    @PrePersist
    void onCreate() {
        createdAt = Instant.now();
    }
}

/**
 * Append-only audit trail of everything that happens in a project: feedback received, revisions
 * documented, status changes, task changes. Every column is updatable=false and the services expose
 * no update or delete, so earlier records are never altered or removed.
 */
@Getter
@Setter
@Entity
@Table(name = "history_entries")
public class HistoryEntry {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "project_id", nullable = false, updatable = false)
    private Project project;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "component_id", updatable = false)
    private ProjectComponent component_id;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "feedback_id", updatable = false)
    private Feedback feedback_id;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "task_id", updatable = false)
    private RevisionTask task_id;

    @Enumerated(EnumType.STRING)
    @Column(length = 20, nullable = false, updatable = false)
    private HistoryType type;

    @Column(columnDefinition = "TEXT", nullable = false, updatable = false)
    private String message;

    /** For status changes: the value before and after (e.g. "Pending" -> "Submitted"). */
    @Column(updatable = false)
    private String oldValue;

    @Column(updatable = false)
    private String newValue;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "actor_id", updatable = false)
    private User actor_id;

    @Column(updatable = false)
    private String actorName;

    @Column(updatable = false)
    private Instant createdAt;

    @PrePersist
    void onCreate() {
        createdAt = Instant.now();
    }
}
